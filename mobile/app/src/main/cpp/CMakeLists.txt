cmake_minimum_required(VERSION 3.22.1)
project(audiotext_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(WHISPER_CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/whisper.cpp")
if (NOT EXISTS "${WHISPER_CORE}/CMakeLists.txt")
    message(FATAL_ERROR "whisper.cpp missing: clone tag v1.7.6 to mobile/vendor/whisper.cpp")
endif()
add_subdirectory("${WHISPER_CORE}" "${CMAKE_CURRENT_BINARY_DIR}/whisper-core")
add_library(audiotext_jni SHARED native_whisper.cpp)
target_include_directories(audiotext_jni PRIVATE "${WHISPER_CORE}/include")
target_link_libraries(audiotext_jni PRIVATE whisper log)
